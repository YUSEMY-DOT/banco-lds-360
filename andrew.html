function renderTabla(lista) { const tbody = document.getElementById("tablaMovimientos"); if (!tbody) return; tbody.innerHTML = ""; const totalCount = Array.isArray(lista) ? lista.length : 0; const elCant = document.getElementById("cantMovimientos"); if (elCant) elCant.textContent = String(totalCount); if (!lista || !lista.length) { tbody.innerHTML = 'NO HAY MOVIMIENTOS REGISTRADOS.'; return; } lista.forEach(m => { if (!m) return; const tr = document.createElement("tr"); const cod = normalizarCodigo(getCodigo(m)); const resp = normalizarCodigo(getResponsable(m)); const concStr = String(getConcepto(m)).toUpperCase(); const monto = getMonto(m); // 1. FECHA (ARRIBA) Y HORA (ABAJO) let fechaOnly = "-"; let horaOnly = ""; const fechaOriginal = getFecha(m); if (fechaOriginal && fechaOriginal !== "-") { const partesFecha = String(fechaOriginal).trim().split(" "); if (partesFecha.length >= 1) fechaOnly = partesFecha[0]; if (partesFecha.length >= 2) { const hPartes = partesFecha[1].split(":"); if (hPartes.length >= 2) { horaOnly = hPartes[0] + ":" + hPartes[1]; // hh:mm } } } // 2. GRADO / ROL ABREVIADO let gradoRaw = String(getGrado(m) || "CAJA GENERAL").toUpperCase(); let gradoCorto = gradoRaw; if (gradoRaw.includes("PROMOTORA")) gradoCorto = "PROM"; else if (gradoRaw.includes("DOCENTE")) gradoCorto = "DOC"; else if (gradoRaw.includes("ADMINISTRACIÓN") || gradoRaw.includes("ADMIN")) gradoCorto = "ADM"; else if (gradoRaw.includes("ESTUDIANTE")) gradoCorto = "EST"; // 3. LIMPIEZA DE OBSERVACIÓN let obsRaw = String(getObservacion(m)).toUpperCase(); let obsLimpia = obsRaw .replace(/^TRANSFERENCIA DE FONDO A [^|]+\|\s*/i, "") .replace(/^DEVOLUCIÓN DE SALDO DE [^|]+\|\s*/i, "") .replace(/^DEVOLUCIÓN DE FONDO A [^|]+\|\s*/i, "") .trim(); if (!obsLimpia) obsLimpia = "-"; let esEgreso = false; let tipoMostrar = "INGRESO"; if (concStr.includes("GERENTE") || concStr.includes("CAPITAL")) { esEgreso = false; tipoMostrar = "INGRESO"; } else if (resp === "LDS-DOC-005" || concStr.includes("FONDO PARA PAGOS")) { esEgreso = true; tipoMostrar = "EGRESO"; } // 4. CUENTA / DESTINO EN 2 LÍNEAS (Nombre Arriba, Código Abajo) let nombreCuenta = getEstudiante(m); let codigoCuenta = cod; if (MAPA_USUARIOS_OFICIALES[cod] && cod !== "LDS-DOC-005") { nombreCuenta = MAPA_USUARIOS_OFICIALES[cod].nombre; } else if (cod === "LDS-DOC-005") { nombreCuenta = "MISTER ANDREW"; } // Abreviar nombres largos para que quepan bien arriba if (nombreCuenta.length > 22) { nombreCuenta = nombreCuenta.split(" ")[0] + " " + (nombreCuenta.split(" ")[1] || ""); } // 5. RESPONSABLE EN 2 LÍNEAS (Ej: MISTER / ANDREW o MISS / LILY) let respNombreOriginal = getResponsable(m) || "SISTEMA"; if (respNombreOriginal.toUpperCase().includes("ANDREW")) { respNombreOriginal = "MISTER ANDREW"; } let partesResp = respNombreOriginal.split(" "); let respArriba = partesResp[0] || "SISTEMA"; let respAbajo = partesResp.slice(1).join(" ") || ""; tr.innerHTML = `
${fechaOnly}
${horaOnly ? `
${horaOnly}
` : ''}
${nombreCuenta}
${codigoCuenta}
${gradoCorto} ${tipoMostrar} ${concStr} ${formatearLDS(monto)}
${respArriba}
${respAbajo ? `
${respAbajo}
` : ''} ${obsLimpia} `; tbody.appendChild(tr); }); }

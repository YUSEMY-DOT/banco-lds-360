function doGet(e) { var params = e.parameter; var callback = params.callback; var accion = (params.accion || "").trim(); var result = {}; try { var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) { ss = SpreadsheetApp.openById("1j0A6XtLbCXJLkVTBXRpK-GnV3QY4qYQ_SOReARpu9ag"); } if (accion === "estudiante" || accion === "CONSULTAR") { result = obtenerEstudianteOUsuario(ss, params.codigo); } else if (accion === "registrarMovimiento" || accion === "REGISTRAR_TRANSACCION") { result = registrarMovimiento(ss, params); } else if (accion === "ADMINISTRADOR") { result = verificarAdministrador(params.clave); } else { result = { ok: false, success: false, mensaje: "Acción no reconocida: " + accion }; } } catch (err) { result = { ok: false, success: false, mensaje: "Error en el servidor: " + err.toString() }; } var jsonOutput = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + "(" + jsonOutput + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonOutput) .setMimeType(ContentService.MimeType.JSON); } } function obtenerUrlImagenDirecta(urlOriginal) { if (!urlOriginal || typeof urlOriginal !== "string") { return "https://via.placeholder.com/150"; } urlOriginal = urlOriginal.trim(); if (urlOriginal.indexOf("drive.google.com") !== -1 || urlOriginal.indexOf("docs.google.com") !== -1) { var match = urlOriginal.match(/\/d\/(.*?)\//) || urlOriginal.match(/id=([^&]*)/); if (match && match[1]) { return "https://lh3.googleusercontent.com/d/" + match[1]; } } if (urlOriginal.indexOf("http") === 0) { return urlOriginal; } return "https://via.placeholder.com/150"; } function obtenerEstudianteOUsuario(ss, codigo) { if (!codigo) return { ok: false, success: false, mensaje: "Código no proporcionado" }; codigo = codigo.toString().trim().toUpperCase(); var sheetEst = ss.getSheetByName("ESTUDIANTES"); var lastRowEst = sheetEst.getLastRow(); if (lastRowEst < 2) return { ok: false, success: false, mensaje: "No hay estudiantes registrados" }; // Cargamos en memoria de un solo golpe para velocidad máxima var dataEst = sheetEst.getRange(1, 1, lastRowEst, 10).getValues(); var usuario = null; var esDocente = codigo.indexOf("LDS-DOC") !== -1; for (var i = 1; i < dataEst.length; i++) { var row = dataEst[i]; var codFila = row[0] ? row[0].toString().trim().toUpperCase() : ""; if (codFila === codigo) { var rolUpper = (row[3] || "").toString().toUpperCase(); if (esDocente || rolUpper.indexOf("DOCENTE") !== -1 || rolUpper.indexOf("PROMOTORA") !== -1 || rolUpper.indexOf("COORDINADORA") !== -1) { esDocente = true; } var rawFoto = row[9] || row[8] || ""; usuario = { codigo: codFila, nombre: row[1] || "", apellido: row[2] || "", rol: row[3] || (esDocente ? "DOCENTE" : "ALUMNO"), grado: row[4] || "", seccion: row[5] || "", telefono: row[6] || "", nombreCompleto: row[7] || (row[1] + " " + row[2]), foto: rawFoto, fotoUrl: obtenerUrlImagenDirecta(rawFoto), esDocente: esDocente }; break; } } if (!usuario) { return { ok: false, success: false, mensaje: "Código no encontrado: " + codigo }; } var sheetMov = ss.getSheetByName("MOVIMIENTOS"); var lastRowMov = sheetMov.getLastRow(); var movimientos = []; var totalIngresos = 0; var totalEgresos = 0; if (lastRowMov >= 2) { var dataMov = sheetMov.getRange(1, 1, lastRowMov, 9).getValues(); for (var j = 1; j < dataMov.length; j++) { var r = dataMov[j]; var codEst = r[1] ? r[1].toString().trim().toUpperCase() : ""; var tipo = r[4] ? r[4].toString().trim().toUpperCase() : ""; var monto = parseFloat(r[6]) || 0; if (codEst === codigo) { if (tipo.indexOf("INGRESO") !== -1 || tipo.includes("BONIF")) { totalIngresos += monto; } else if (tipo.indexOf("EGRESO") !== -1 || tipo.indexOf("GASTO") !== -1) { totalEgresos += monto; } movimientos.push({ fechaTexto: r[0] ? r[0].toString() : "", codigo: r[1] ? r[1].toString() : "", estudiante: r[2] ? r[2].toString() : "", grado: r[3] ? r[3].toString() : "", tipo: r[4] ? r[4].toString() : "", concepto: r[5] ? r[5].toString() : "", monto: monto, responsable: r[7] ? r[7].toString() : "", observacion: r[8] ? r[8].toString() : "" }); } } } var saldoActual = totalIngresos - totalEgresos; return { ok: true, success: true, estudiante: usuario, saldoActual: saldoActual, cuenta: { totalIngresos: totalIngresos, totalEgresos: totalEgresos, saldoActual: saldoActual }, movimientos: movimientos }; } function registrarMovimiento(ss, params) { var codigo = params.codigo ? params.codigo.toString().trim().toUpperCase() : ""; var tipo = params.tipo ? params.tipo.toString().trim().toUpperCase() : "EGRESO"; var concepto = params.concepto || ""; var monto = parseFloat(params.monto) || 0; var responsable = params.responsable || "DOCENTE"; var observacion = params.observacion || ""; if (!codigo || monto <= 0) return { ok: false, success: false, mensaje: "Datos incompletos o monto inválido" }; var sheetEst = ss.getSheetByName("ESTUDIANTES"); var lastRowEst = sheetEst.getLastRow(); var dataEst = sheetEst.getRange(1, 1, lastRowEst, 8).getValues(); var nombreCompleto = "", grado = ""; for (var i = 1; i < dataEst.length; i++) { if (dataEst[i][0] && dataEst[i][0].toString().trim().toUpperCase() === codigo) { nombreCompleto = dataEst[i][7] || (dataEst[i][1] + " " + dataEst[i][2]); grado = dataEst[i][4] || ""; break; } } var sheetMov = ss.getSheetByName("MOVIMIENTOS"); var timestamp = Utilities.formatDate(new Date(), "America/Lima", "dd/MM/yyyy HH:mm:ss"); sheetMov.appendRow([timestamp, codigo, nombreCompleto, grado, tipo, concepto, monto, responsable, observacion]); return { ok: true, success: true, mensaje: "Registrado con éxito" }; } function verificarAdministrador(clave) { var esAdmin = clave && clave.toString().trim().toUpperCase() === "360LDS"; return { ok: esAdmin, success: true, autorizado: esAdmin }; }

function doGet(e) {
  return procesarPeticion(e);
}

function doPost(e) {
  return procesarPeticion(e);
}

function procesarPeticion(e) {
  var params = e ? e.parameter : {};
  var callback = params.callback;
  var accion = params.accion ? params.accion.toString().trim().toUpperCase() : "";
  var result = {};

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      ss = SpreadsheetApp.openById("1Zr_2FGD7x1T5ppfLoq_oOa6S4-CdHRU9DIr92x-fs5A");
    }

    switch (accion) {
      case "CONSULTAR":
      case "ESTUDIANTE":
        result = obtenerEstudianteOUsuario(ss, params.codigo);
        break;
      case "ESTUDIANTES":
      case "LISTAESTUDIANTES":
        result = obtenerListaEstudiantes(ss);
        break;
      case "MISES":
      case "LISTAMISES":
        result = obtenerListaMises(ss);
        break;
      case "MOVIMIENTOS":
      case "LISTAMOVIMIENTOS":
      case "OBTENER_MOVIMIENTOS":
        result = obtenerTodosLosMovimientos(ss);
        break;
      case "LOGINALUMNO":
        result = validarLogin(ss, params.codigo, params.clave);
        break;
      case "REGISTRAR_TRANSACCION":
      case "REGISTRARMOVIMIENTO":
        result = registrarMovimiento(ss, params);
        break;
      case "PRESTAMO":
      case "REGISTRAR_PRESTAMO":
        result = registrarPrestamo(ss, params);
        break;
      case "INCIDENCIA":
      case "REGISTRAR_INCIDENCIA":
        result = registrarIncidencia(ss, params);
        break;
      case "RECORDATORIO":
      case "ENVIAR_RECORDATORIO":
        result = registrarRecordatorio(ss, params);
        break;
      case "CONFIGOPCIONES":
        result = obtenerConfigOpciones(ss);
        break;
      case "CUENTAS":
      case "OBTENER_CUENTAS":
        result = obtenerTodasLasCuentas(ss);
        break;
      default:
        if (params.codigo) {
          result = obtenerEstudianteOUsuario(ss, params.codigo);
        } else {
          result = obtenerListaEstudiantes(ss);
        }
        break;
    }
  } catch (err) {
    result = { ok: false, success: false, mensaje: "Error en el servidor: " + err.toString() };
  }

  var jsonOutput = JSON.stringify(result);
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + jsonOutput + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  } else {
    return ContentService.createTextOutput(jsonOutput)
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function obtenerHojaUsuarios(ss) {
  return ss.getSheetByName("USUARIOS") || ss.getSheetByName("ESTUDIANTES");
}

function parseMontoVal(val) {
  if (typeof val === "number") return Math.trunc(val);
  if (!val) return 0;
  var str = val.toString().replace(/LDS/gi, "").trim();
  if (str.indexOf(".") !== -1 && str.indexOf(",") === -1) {
    var partes = str.split(".");
    if (partes.length > 1 && partes[partes.length - 1].length === 3) {
      str = str.replace(/\./g, "");
    }
  }
  str = str.replace(/,/g, ".");
  var n = parseFloat(str);
  return isNaN(n) ? 0 : Math.trunc(n);
}

function obtenerTodosLosMovimientos(ss) {
  var sheetMov = ss.getSheetByName("MOVIMIENTOS");
  if (!sheetMov) return { ok: false, success: false, movimientos: [] };

  var dataMov = sheetMov.getDataRange().getDisplayValues();
  var movimientos = [];

  for (var j = 1; j < dataMov.length; j++) {
    var r = dataMov[j];
    if (!r[0] && !r[1]) continue;

    var rawFecha = r[0] ? r[0].toString().trim() : "";
    var fechaTexto = rawFecha;

    if (rawFecha.indexOf("GMT") !== -1 || rawFecha.indexOf("Mon") !== -1 || rawFecha.indexOf("Tue") !== -1 || rawFecha.indexOf("Wed") !== -1 || rawFecha.indexOf("Thu") !== -1 || rawFecha.indexOf("Fri") !== -1 || rawFecha.indexOf("Sat") !== -1 || rawFecha.indexOf("Sun") !== -1) {
      try {
        var d = new Date(rawFecha);
        if (!isNaN(d.getTime())) {
          fechaTexto = Utilities.formatDate(d, "America/Lima", "dd/MM/yyyy HH:mm:ss");
        } else {
          fechaTexto = rawFecha.split("GMT")[0].trim();
        }
      } catch(e) {
        fechaTexto = rawFecha.split("GMT")[0].trim();
      }
    }

    movimientos.push({
      fechaTexto: fechaTexto,
      codigo: r[1] ? r[1].toString().trim() : "",
      estudiante: r[2] ? r[2].toString().trim() : "",
      grado: r[3] ? r[3].toString().trim() : "",
      tipo: r[4] ? r[4].toString().trim() : "",
      concepto: r[5] ? r[5].toString().trim() : "",
      monto: parseMontoVal(r[6]),
      responsable: r[7] ? r[7].toString().trim() : "",
      observacion: r[8] ? r[8].toString().trim() : "",
      categoria: r[9] ? r[9].toString().trim() : "",
      estado: r[11] ? r[11].toString().trim() : ""
    });
  }

  movimientos.reverse();
  return { ok: true, success: true, movimientos: movimientos };
}

function obtenerListaEstudiantes(ss) {
  var sheetEst = obtenerHojaUsuarios(ss);
  if (!sheetEst) return { ok: false, success: false, estudiantes: [] };

  var dataEst = sheetEst.getDataRange().getValues();
  var estudiantes = [];

  for (var i = 1; i < dataEst.length; i++) {
    var row = dataEst[i];
    var cod = row[0] ? row[0].toString().trim().toUpperCase() : "";
    if (!cod) continue;

    var nom = row[1] ? row[1].toString().trim() : "";
    var ape = row[2] ? row[2].toString().trim() : "";
    var rol = row[3] ? row[3].toString().trim() : "ALUMNO";
    var grado = row[4] ? row[4].toString().trim() : "";
    var seccion = row[5] ? row[5].toString().trim() : "";
    var nombreCompleto = row[7] ? row[7].toString().trim() : (nom + (ape ? " " + ape : ""));
    var foto = row[9] || row[8] || "";

    if (foto.toString().toUpperCase().indexOf("SIN_FOTO") !== -1) foto = "";

    var rolUpper = rol.toUpperCase();
    if (cod.indexOf("LDS-DOC") === -1 && rolUpper.indexOf("DOCENTE") === -1 && rolUpper.indexOf("GERENTE") === -1 && rolUpper.indexOf("COORDINADORA") === -1) {
      estudiantes.push({
        codigo: cod,
        nombre: nom,
        apellido: ape,
        nombreCompleto: nombreCompleto,
        rol: rol,
        grado: grado,
        seccion: seccion,
        fotoUrl: foto,
        foto: foto
      });
    }
  }

  return { ok: true, success: true, estudiantes: estudiantes };
}

}

function obtenerEstudianteOUsuario(ss, codigo) {
  if (!codigo) return { ok: false, success: false, mensaje: "Código no proporcionado" };
  codigo = codigo.toString().trim().toUpperCase();

  var sheetEst = obtenerHojaUsuarios(ss);
  if (!sheetEst) return { ok: false, success: false, mensaje: "Hoja de usuarios no encontrada" };

  var dataEst = sheetEst.getDataRange().getValues();
  var usuario = null;
  var esDocente = codigo.indexOf("LDS-DOC") !== -1;

  for (var i = 1; i < dataEst.length; i++) {
    var row = dataEst[i];
    if (row[0] && row[0].toString().trim().toUpperCase() === codigo) {
      var rolUpper = (row[3] || "").toString().toUpperCase();
      if (codigo.indexOf("LDS-DOC") !== -1 || rolUpper.indexOf("DOCENTE") !== -1 || rolUpper.indexOf("PROMOTORA") !== -1 || rolUpper.indexOf("COORDINADORA") !== -1 || rolUpper.indexOf("GERENTE") !== -1) {
        esDocente = true;
      }
      var fotoRaw = row[9] || row[8] || "";
      if (fotoRaw.toString().toUpperCase().indexOf("SIN_FOTO") !== -1) fotoRaw = "";

      usuario = {
        codigo: row[0].toString().trim(),
        nombre: row[1] || "",
        apellido: row[2] || "",
        rol: row[3] || (esDocente ? "DOCENTE" : "ALUMNO"),
        grado: row[4] || "",
        seccion: row[5] || "",
        nombreCompleto: row[7] || (row[1] + " " + (row[2] || "")),
        foto: fotoRaw,
        fotoUrl: fotoRaw,
        esDocente: esDocente
      };
      break;
    }
  }

  if (!usuario) return { ok: false, success: false, mensaje: "Código no encontrado: " + codigo };

  var sheetMov = ss.getSheetByName("MOVIMIENTOS");
  if (!sheetMov) return { ok: false, success: false, mensaje: "Hoja MOVIMIENTOS no encontrada" };

  var dataMov = sheetMov.getDataRange().getDisplayValues();
  var movimientos = [];
  var totalIngresosAlumno = 0;
  var totalEgresosAlumno = 0;
  var fondoRecibidoMaestra = 0;
  var totalPagadoMaestraAAlumnos = 0;
  var totalCobradoIngresadoMaestra = 0;

  var nombreDocenteUpper = (usuario.nombreCompleto || "").toUpperCase();
  var nombreSimpleUpper = (usuario.nombre || "").toUpperCase();
  var palabrasDocente = nombreDocenteUpper.split(" ").filter(function(w) { return w.length > 2; });

  for (var j = 1; j < dataMov.length; j++) {
    var r = dataMov[j];
    if (!r[0] && !r[1]) continue;

    var rawFecha = r[0] ? r[0].toString().trim() : "";
    var fechaTexto = rawFecha;

    if (rawFecha.indexOf("GMT") !== -1 || rawFecha.indexOf("Mon") !== -1 || rawFecha.indexOf("Tue") !== -1 || rawFecha.indexOf("Wed") !== -1 || rawFecha.indexOf("Thu") !== -1 || rawFecha.indexOf("Fri") !== -1 || rawFecha.indexOf("Sat") !== -1 || rawFecha.indexOf("Sun") !== -1) {
      try {
        var d = new Date(rawFecha);
        if (!isNaN(d.getTime())) {
          fechaTexto = Utilities.formatDate(d, "America/Lima", "dd/MM/yyyy HH:mm:ss");
        } else {
          fechaTexto = rawFecha.split("GMT")[0].trim();
        }
      } catch(e) {
        fechaTexto = rawFecha.split("GMT")[0].trim();
      }
    }

    var codEstudiante = r[1] ? r[1].toString().trim().toUpperCase() : "";
    var nomEstudiante = r[2] ? r[2].toString().trim() : "";
    var gradoEstudiante = r[3] ? r[3].toString().trim() : "";
    var tipoRaw = r[4] ? r[4].toString().trim().toUpperCase() : "";
    var concepto = r[5] ? r[5].toString().trim() : "";
    var monto = parseMontoVal(r[6]);
    var responsable = r[7] ? r[7].toString().trim().toUpperCase() : "";
    var observacion = r[8] ? r[8].toString().trim() : "";

    if (observacion.indexOf("ID_OPERACION") !== -1 || observacion.indexOf("BOTON") !== -1) {
      observacion = observacion.replace(/BOTON\s*\|\s*ID_OPERACION=[^\s]+/gi, "").trim();
    }

    var concUpper = concepto.toUpperCase();
    var esCobroOSancion = (
      tipoRaw === "COBRO" || tipoRaw === "COBRAR" ||
      concUpper.indexOf("MULTA") !== -1 || concUpper.indexOf("SANCIÓN") !== -1 ||
      concUpper.indexOf("INCIDENCIA") !== -1 || concUpper.indexOf("VERDAD") !== -1 ||
      concUpper.indexOf("TIENDA") !== -1 || concUpper.indexOf("DESORDEN") !== -1 ||
      concUpper.indexOf("TARDANZA") !== -1 || concUpper.indexOf("UNIFORME") !== -1
    );

    if (!esDocente && (codEstudiante === codigo || (nomEstudiante && nomEstudiante.toUpperCase() === nombreDocenteUpper))) {
      if (esCobroOSancion || tipoRaw === "EGRESO") {
        totalEgresosAlumno += monto;
        movimientos.push({
          fechaTexto: fechaTexto, codigo: codEstudiante, estudiante: nomEstudiante, grado: gradoEstudiante,
          tipo: "EGRESO", concepto: concepto, monto: monto, responsable: r[7] ? r[7].toString() : "", observacion: observacion
        });
      } else {
        totalIngresosAlumno += monto;
        movimientos.push({
          fechaTexto: fechaTexto, codigo: codEstudiante, estudiante: nomEstudiante, grado: gradoEstudiante,
          tipo: "INGRESO", concepto: concepto, monto: monto, responsable: r[7] ? r[7].toString() : "", observacion: observacion
        });
      }
    }

    if (esDocente) {
      var esFondoAsignadoADocente = (codEstudiante === codigo);
      var esCoincidenciaResponsable = (
        responsable.indexOf(codigo) !== -1 ||
        (nombreDocenteUpper && responsable.indexOf(nombreDocenteUpper) !== -1) ||
        (nombreSimpleUpper && responsable.indexOf(nombreSimpleUpper) !== -1)
      );

      if (!esCoincidenciaResponsable && palabrasDocente.length >= 2) {
        var coincidentes = 0;
        for (var p = 0; p < palabrasDocente.length; p++) {
          if (responsable.indexOf(palabrasDocente[p]) !== -1) coincidentes++;
        }
        if (coincidentes >= 2) esCoincidenciaResponsable = true;
      }

      if (esFondoAsignadoADocente) {
        fondoRecibidoMaestra += monto;
        movimientos.push({
          fechaTexto: fechaTexto, codigo: codEstudiante, estudiante: nomEstudiante, grado: gradoEstudiante,
          tipo: "INGRESO", concepto: concepto, monto: monto, responsable: r[7] ? r[7].toString() : "", observacion: observacion
        });
      } else if (esCoincidenciaResponsable && codEstudiante !== codigo) {
        if (tipoRaw === "RECORDATORIO") {
          movimientos.push({
            fechaTexto: fechaTexto, codigo: codEstudiante, estudiante: nomEstudiante, grado: gradoEstudiante,
            tipo: "RECORDATORIO", concepto: concepto, monto: monto, responsable: r[7] ? r[7].toString() : "", observacion: observacion
          });
        } else if (esCobroOSancion) {
          totalCobradoIngresadoMaestra += monto;
          movimientos.push({
            fechaTexto: fechaTexto, codigo: codEstudiante, estudiante: nomEstudiante, grado: gradoEstudiante,
            tipo: "INGRESO", concepto: concepto, monto: monto, responsable: r[7] ? r[7].toString() : "", observacion: observacion
          });
        } else {
          totalPagadoMaestraAAlumnos += monto;
          movimientos.push({
            fechaTexto: fechaTexto, codigo: codEstudiante, estudiante: nomEstudiante, grado: gradoEstudiante,
            tipo: "EGRESO", concepto: concepto, monto: monto, responsable: r[7] ? r[7].toString() : "", observacion: observacion
          });
        }
      }
    }
  }

  movimientos.reverse();
  var saldoActual = 0;
  var totalEntregadoMaestra = fondoRecibidoMaestra;

  if (esDocente) {
    if (fondoRecibidoMaestra === 0) {
      fondoRecibidoMaestra = 15000;
      totalEntregadoMaestra = 15000;
    }
    saldoActual = fondoRecibidoMaestra - totalPagadoMaestraAAlumnos + totalCobradoIngresadoMaestra;
    if (saldoActual < 0) saldoActual = 0;
  } else {
    saldoActual = Math.max(0, totalIngresosAlumno - totalEgresosAlumno);
  }

  return {
    ok: true, success: true, estudiante: usuario,
    cuenta: {
      totalIngresos: totalIngresosAlumno, totalEgresos: totalEgresosAlumno,
      totalEntregadoMaestra: totalEntregadoMaestra, totalPagadoAAlumnos: totalPagadoMaestraAAlumnos,
      totalCobradoAAlumnos: totalCobradoIngresadoMaestra, saldoActual: saldoActual
    },
    movimientos: movimientos
  };
}

function registrarMovimiento(ss, params) {
  var codigo = params.codigo || params.codigoDestino || "";
  codigo = codigo.toString().trim().toUpperCase();

  var tipo = params.tipo ? params.tipo.toString().trim().toUpperCase() : "EGRESO";
  var concepto = params.concepto || "";
  var monto = parseFloat(params.monto) || 0;
  var responsable = params.responsable || "DOCENTE";
  var observacion = params.observacion || "";
  var categoria = params.categoria || "2026";

  if (!codigo) return { ok: false, success: false, mensaje: "Código de cuenta requerido." };
  if (monto <= 0) return { ok: false, success: false, mensaje: "El monto debe ser mayor a 0." };

  var sheetEst = obtenerHojaUsuarios(ss);
  var nombreCompleto = codigo;
  var grado = "";

  if (sheetEst) {
    var dataEst = sheetEst.getDataRange().getValues();
    for (var i = 1; i < dataEst.length; i++) {
      var row = dataEst[i];
      if (row[0] && row[0].toString().trim().toUpperCase() === codigo) {
        nombreCompleto = row[7] || (row[1] + " " + (row[2] || ""));
        grado = row[4] || "";
        break;
      }
    }
  }

  var sheetMov = ss.getSheetByName("MOVIMIENTOS");
  if (!sheetMov) return { ok: false, success: false, mensaje: "No existe la hoja MOVIMIENTOS." };

  var timestamp = Utilities.formatDate(new Date(), "America/Lima", "dd/MM/yyyy HH:mm:ss");
  var montoFormateado = Math.trunc(monto).toLocaleString("de-DE") + " LDS";

  sheetMov.appendRow([
    timestamp, codigo, nombreCompleto, grado, tipo, concepto, montoFormateado, responsable, observacion, categoria, "", "PROCESADO"
  ]);

  actualizarSaldoEnHojaCuentas(ss, codigo);
  return { ok: true, success: true, mensaje: "¡Transacción registrada con éxito!" };
}

function registrarRecordatorio(ss, params) {
  return registrarMovimiento(ss, {
    codigo: params.codigo, tipo: "RECORDATORIO", concepto: "RECORDATORIO: " + (params.servicio || "AVISO"),
    monto: parseFloat(params.monto) || 0, responsable: params.responsable || "DOCENTE",
    observacion: params.observacion || "Aviso", categoria: "AVISOS"
  });
}

function registrarPrestamo(ss, params) {
  return registrarMovimiento(ss, {
    codigo: params.destino || params.codigo, tipo: "INGRESO",
    concepto: params.concepto || "PRÉSTAMO / TRANSFERENCIA DE CAJA",
    monto: parseFloat(params.monto) || 0, responsable: params.origen || params.responsable || "CAJA",
    observacion: params.observacion || "Transferencia", categoria: "TRANSFERENCIA"
  });
}

function registrarIncidencia(ss, params) {
  return registrarMovimiento(ss, {
    codigo: params.codigo, tipo: "COBRO", concepto: params.falta || "INCIDENCIA",
    monto: parseFloat(params.monto) || 0, responsable: params.responsable || "DOCENTE",
    observacion: "SANCIÓN: " + (params.observacion || ""), categoria: "SANCIONES_Y_MULTAS"
  });
}

function actualizarSaldoEnHojaCuentas(ss, codigo) {
  var sheetCuentas = ss.getSheetByName("CUENTAS");
  if (!sheetCuentas) return;
  var dataCuentas = sheetCuentas.getDataRange().getValues();
  var filaEncontrada = -1;

  for (var i = 1; i < dataCuentas.length; i++) {
    if (dataCuentas[i][0] && dataCuentas[i][0].toString().trim().toUpperCase() === codigo) {
      filaEncontrada = i + 1;
      break;
    }
  }
  if (filaEncontrada === -1) return;

  var sheetMov = ss.getSheetByName("MOVIMIENTOS");
  if (!sheetMov) return;
  var dataMov = sheetMov.getDataRange().getDisplayValues();
  var totalIngresos = 0;
  var totalEgresos = 0;

  for (var j = 1; j < dataMov.length; j++) {
    var r = dataMov[j];
    if (r[1] && r[1].toString().trim().toUpperCase() === codigo) {
      var tipo = r[4] ? r[4].toString().trim().toUpperCase() : "EGRESO";
      var monto = parseMontoVal(r[6]);
      if (tipo.indexOf("INGRESO") !== -1) totalIngresos += monto;
      else if (tipo.indexOf("EGRESO") !== -1 || tipo.indexOf("COBRO") !== -1) totalEgresos += monto;
    }
  }

  var saldoActual = Math.max(0, totalIngresos - totalEgresos);
  sheetCuentas.getRange(filaEncontrada, 5).setValue(Math.trunc(totalIngresos).toLocaleString("de-DE") + " LDS");
  sheetCuentas.getRange(filaEncontrada, 6).setValue(Math.trunc(totalEgresos).toLocaleString("de-DE") + " LDS");
  sheetCuentas.getRange(filaEncontrada, 7).setValue(Math.trunc(saldoActual).toLocaleString("de-DE") + " LDS");
}

function validarLogin(ss, codigo, clave) {
  if (!codigo) return { ok: false, success: false, mensaje: "Código no proporcionado" };
  codigo = codigo.toString().trim().toUpperCase();
  var sheetCuentas = ss.getSheetByName("CUENTAS") || obtenerHojaUsuarios(ss);
  if (!sheetCuentas) return { ok: false, success: false, mensaje: "Hoja no encontrada" };

  var data = sheetCuentas.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] && data[i][0].toString().trim().toUpperCase() === codigo) {
      var claveGuardada = data[i][2] ? data[i][2].toString().trim() : "";
      if (!claveGuardada || claveGuardada === clave) return { ok: true, success: true, mensaje: "Acceso correcto" };
    }
  }
  return { ok: false, success: false, mensaje: "Credenciales incorrectas" };
}

function obtenerConfigOpciones(ss) {
  var sheetConfig = ss ? ss.getSheetByName("CONFIG_OPCIONES") : null;
  if (!sheetConfig) {
    return {
      ok: true, success: true,
      opciones: {
        "INGRESOS_Y_SUELDOS": ["Ingreso Sueldo Base Mensual", "Ingreso Sueldo por Rol de Liderazgo", "Ingreso Bono por desempeño"],
        "PREMIOS_Y_BONIFICACIONES": ["Bono por llevar el uniforme impecable y reglamentario", "Bono Participación destacada en clase"],
        "AHORROS_E_INVERSIONES": ["Depósito a cuenta de ahorros (Ahorro voluntario)", "Inversión en fondo común del aula / cooperativa"],
        "PAGOS_Y_GASTOS": ["Pago Compra en la tienda escolar / cafetín", "Pago de alquiler / mantenimiento de carpeta"],
        "SANCIONES_Y_MULTAS": ["Multa por llevar mal el uniforme o incompleto", "Incumplimiento de tareas o asignaciones"],
        "COSTOS_Y_OTROS": ["Impuesto escolar mensual por uso de instalaciones", "Transacción personalizada / Ajuste de caja"]
      }
    };
  }

  var data = sheetConfig.getDataRange().getValues();
  var headers = data[0];
  var opciones = {};

  for (var col = 0; col < headers.length; col++) {
    var catKey = headers[col] ? headers[col].toString().trim() : "";
    if (!catKey) continue;
    opciones[catKey] = [];
    for (var row = 1; row < data.length; row++) {
      var val = data[row][col] ? data[row][col].toString().trim() : "";
      if (val) opciones[catKey].push(val);
    }
  }
  return { ok: true, success: true, opciones: opciones };
}

function obtenerTodasLasCuentas(ss) {
  var sheetCuentas = ss.getSheetByName("CUENTAS");
  if (!sheetCuentas) return { ok: false, success: false, cuentas: [] };
  var data = sheetCuentas.getDataRange().getValues();
  var cuentas = [];

  for (var i = 1; i < data.length; i++) {
    if (!data[i][0]) continue;
    cuentas.push({
      codigo: data[i][0].toString().trim(),
      usuario: data[i][1] || "",
      grado: data[i][3] || "",
      totalIngresos: data[i][4] || "0 LDS",
      totalEgresos: data[i][5] || "0 LDS",
      saldoActual: data[i][6] || "0 LDS",
      activo: data[i][7] || "SI"
    });
  }
  return { ok: true, success: true, cuentas: cuentas };
}
